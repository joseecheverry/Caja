        Código para pegar en Apps Script 
        // Extensiones ▸ Apps Script en tu hoja. Borra todo, pega esto y guarda.
// Implementar ▸ Nueva implementación ▸ tipo: Aplicación web
// Ejecutar como: Yo  ·  Quién tiene acceso: CUALQUIER USUARIO
// Cada cambio de código exige una implementación NUEVA (la URL cambia).
// Prueba de lectura:   TU_URL/exec?callback=x   →   x({"movs":[]})

const HOJA = 'Movimientos';
const CABECERA = ['id','libro','fecha','tipo','monto','categoria','medio','nota','creado'];

function hoja_(){
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let h = ss.getSheetByName(HOJA);
  if(!h) h = ss.insertSheet(HOJA);
  if(h.getLastRow() === 0) h.appendRow(CABECERA);
  return h;
}

function salida_(txt, cb){
  if(cb){
    return ContentService.createTextOutput(cb + '(' + txt + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(txt)
    .setMimeType(ContentService.MimeType.JSON);
}

function leer_(){
  const v = hoja_().getDataRange().getValues();
  const cab = v.shift() || [];
  return v.map(function(f){
    const o = {};
    cab.forEach(function(c, i){ o[c] = f[i]; });
    if(o.fecha instanceof Date){
      o.fecha = Utilities.formatDate(o.fecha, 'America/Bogota', 'yyyy-MM-dd');
    }
    o.monto = Number(o.monto) || 0;
    return o;
  });
}

function procesar_(p){
  const h = hoja_();
  const v = h.getDataRange().getValues();
  let n = 0;

  if(p.accion === 'borrar'){
    for(let i = v.length - 1; i >= 1; i--){
      if(String(v[i][0]) === String(p.id)){ h.deleteRow(i + 1); n++; }
    }
    SpreadsheetApp.flush();
    return {ok: true, borrados: n};
  }

  (p.movs || []).forEach(function(m){
    const fila = [m.id, m.libro, m.fecha, m.tipo, m.monto,
                  m.categoria, m.medio, m.nota, new Date()];
    let pos = -1;
    for(let i = 1; i < v.length; i++){
      if(String(v[i][0]) === String(m.id)){ pos = i + 1; break; }
    }
    if(pos > 0) h.getRange(pos, 1, 1, fila.length).setValues([fila]);
    else h.appendRow(fila);
    n++;
  });
  SpreadsheetApp.flush();
  return {ok: true, guardados: n};
}

function doGet(e){
  const p = (e && e.parameter) || {};
  if(p.datos){
    const r = procesar_(JSON.parse(p.datos));
    r.v = 3;
    return salida_(JSON.stringify(r), p.callback);
  }
  return salida_(JSON.stringify({v: 3, movs: leer_()}), p.callback);
}

function doPost(e){
  const p = JSON.parse(e.postData.contents);
  const r = procesar_(p);
  r.v = 3;
  return salida_(JSON.stringify(r), null);
}
